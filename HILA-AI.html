import express from "express"; import cors from "cors"; import "dotenv/config"; import { GoogleGenAI } from "@google/genai"; const app = express(); app.use(cors()); app.use(express.json({ limit: "1mb" })); // إعداد Gemini const apiKey = process.env.GEMINI_API_KEY; const client = apiKey ? new GoogleGenAI({ apiKey }) : null; const PRIMARY_MODEL = process.env.GEMINI_MODEL || "gemini-3.8-flash"; const FALLBACK_MODEL = process.env.GEMINI_FALLBACK_MODEL || "gemini-3.6-flash"; const SYSTEM_PROMPT = ` أنت HILA AI، مساعد ذكي مفيد وواضح. أجب بلغة المستخدم. إذا كتب المستخدم بالدارجة الجزائرية، أجب بالدارجة الجزائرية. كن ودودًا، اشرح ببساطة، ولا تخترع معلومات. `; // الصفحة الرئيسية app.get("/", (_req, res) => { res.status(200).send("HILA AI backend is running."); }); // فحص حالة السيرفر app.get("/health", (_req, res) => { res.status(200).json({ status: "ok", service: "HILA AI", model: PRIMARY_MODEL, }); }); // انتظار بين محاولات الاتصال const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms)); // استخراج كود الخطأ function getStatus(error) { return Number( error?.status || error?.code || error?.response?.status || 0 ); } // تحديد الأخطاء المؤقتة function isRetryable(status) { return [408, 429, 500, 502, 503, 504].includes(status); } // طلب Gemini مع إعادة المحاولة async function generateWithRetry(model, contents, retries = 3) { let lastError; for (let attempt = 0; attempt <= retries; attempt++) { try { const result = await client.models.generateContent({ model, contents, config: { systemInstruction: SYSTEM_PROMPT, temperature: 0.7, maxOutputTokens: 2048, }, }); const text = result?.text; if (typeof text !== "string" || !text.trim()) { throw new Error("Gemini returned an empty response."); } return text.trim(); } catch (error) { lastError = error; const status = getStatus(error); // لا نكرر الأخطاء غير المؤقتة if (!isRetryable(status) || attempt === retries) { throw error; } // انتظار تدريجي: 1، 2، 4 ثوانٍ const delay = Math.min(1000 * 2 ** attempt, 8000); console.log( `[${model}] Error ${status || "unknown"} - retry ${ attempt + 1 }/${retries} after ${delay}ms` ); await sleep(delay); } } throw lastError; } // واجهة المحادثة app.post("/api/chat", async (req, res) => { try { // التأكد من وجود المفتاح if (!client) { return res.status(500).json({ error: "مفتاح Gemini API غير مضبوط في إعدادات Render Environment.", }); } // استقبال الرسائل من الواجهة const incoming = Array.isArray(req.body?.messages) ? req.body.messages : []; // تنظيف الرسائل وتحويلها لصيغة Gemini const messages = incoming .filter( (m) => m && (m.role === "user" || m.role === "assistant") && typeof m.content === "string" && m.content.trim().length > 0 ) .slice(-30) .map((m) => ({ role: m.role === "assistant" ? "model" : "user", parts: [ { text: m.content.slice(0, 8000), }, ], })); // التحقق من الرسالة الأخيرة if ( messages.length === 0 || messages[messages.length - 1].role !== "user" ) { return res.status(400).json({ error: "اكتب رسالة أولاً.", }); } let reply; // محاولة الموديل الرئيسي try { console.log(`Using primary model: ${PRIMARY_MODEL}`); reply = await generateWithRetry( PRIMARY_MODEL, messages, 3 ); } catch (primaryError) { const primaryStatus = getStatus(primaryError); console.error("Primary model failed:", { model: PRIMARY_MODEL, status: primaryStatus, message: primaryError?.message, }); // عدم الانتقال إلى موديل آخر عند أخطاء المفتاح أو الطلب if ( primaryStatus && !isRetryable(primaryStatus) ) { throw primaryError; } // محاولة الموديل الاحتياطي if ( FALLBACK_MODEL && FALLBACK_MODEL !== PRIMARY_MODEL ) { console.log(`Switching to fallback: ${FALLBACK_MODEL}`); try { reply = await generateWithRetry( FALLBACK_MODEL, messages, 1 ); } catch (fallbackError) { console.error("Fallback model failed:", { model: FALLBACK_MODEL, status: getStatus(fallbackError), message: fallbackError?.message, }); throw fallbackError; } } else { throw primaryError; } } // إرسال الرد إلى الواجهة return res.status(200).json({ reply, }); } catch (error) { const status = getStatus(error); console.error("HILA API error:", { status, message: error?.message, }); // خطأ المفتاح if (status === 401 || status === 403) { return res.status(status).json({ error: "مفتاح Gemini غير صالح أو لا يملك صلاحية الوصول.", }); } // تجاوز الحد أو ضغط الطلبات if (status === 429) { return res.status(429).json({ error: "طلبات كثيرة أو تجاوز الحد المسموح. حاول بعد قليل.", }); } // ضغط أو انقطاع مؤقت if ([408, 500, 502, 503, 504].includes(status) || status === 0) { return res.status(503).json({ error: "Gemini غير متاح مؤقتًا حتى بعد إعادة المحاولة والموديل الاحتياطي.", }); } return res.status(500).json({ error: "حدث خطأ أثناء معالجة طلبك.", }); } }); // التعامل مع المسارات غير الموجودة app.use((req, res) => { res.status(404).json({ error: "المسار غير موجود.", }); }); // معالجة أخطاء JSON أو Express app.use((error, req, res, next) => { console.error("Server middleware error:", error.message); if (res.headersSent) { return next(error); } res.status(400).json({ error: "طلب غير صالح.", }); }); // تشغيل السيرفر على Render const port = Number(process.env.PORT || 3001); app.listen(port, "0.0.0.0", () => { console.log("================================"); console.log("HILA AI backend started"); console.log(`Port: ${port}`); console.log(`Primary model: ${PRIMARY_MODEL}`); console.log(`Fallback model: ${FALLBACK_MODEL}`); console.log("================================"); });
