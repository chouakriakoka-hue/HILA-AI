// Gemini backend const API_URL = "/api/chat"; const messages = []; const box = document.getElementById("messages"); const input = document.getElementById("prompt"); const sendButton = document.getElementById("send"); const welcome = document.getElementById("welcome"); function addMessage(role, content, extraClass = "") { const el = document.createElement("div"); el.className = `msg ${role} ${extraClass}`; el.textContent = content; box.appendChild(el); el.scrollIntoView({ behavior: "smooth", block: "end" }); return el; } async function sendMessage(text) { const value = text.trim(); if (!value || sendButton.disabled) return; welcome.style.display = "none"; addMessage("user", value); messages.push({ role: "user", content: value }); input.value = ""; sendButton.disabled = true; const loading = addMessage("assistant", "HILA AI تفكّر..."); try { const response = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ messages: messages.slice(-30) }) }); const raw = await response.text(); let data; try { data = JSON.parse(raw); } catch { throw new Error( `السيرفر رجّع رد غير صالح (${response.status}). تأكد من Backend.` ); } if (!response.ok) { throw new Error(data.error || `خطأ HTTP ${response.status}`); } const reply = data.reply || "لم يصل رد."; loading.textContent = reply; messages.push({ role: "assistant", content: reply }); } catch (err) { loading.className = "msg error"; loading.textContent = "ما قدرناش نتصلو بالخادم: " + err.message + "\nتأكد من إعداد Gemini API ومن تشغيل الخادم على Render."; } finally { sendButton.disabled = false; input.focus(); } } document.getElementById("form").addEventListener("submit", e => { e.preventDefault(); sendMessage(input.value); }); document.getElementById("newChat").addEventListener("click", () => { messages.length = 0; box.replaceChildren(); welcome.style.display = ""; input.value = ""; }); document.querySelectorAll("[data-prompt]").forEach(btn => btn.addEventListener("click", () => sendMessage(btn.dataset.prompt)) );
